cmake_minimum_required(VERSION 3.24)

project(th06 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "")

set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
add_subdirectory(external/SDL)

set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_VENDORED ON CACHE BOOL "" FORCE)
set(SDL2IMAGE_AVIF OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_WEBP OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_JXL OFF CACHE BOOL "" FORCE)
add_subdirectory(external/SDL_image)

set(SDL2TTF_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2TTF_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2TTF_VENDORED ON CACHE BOOL "" FORCE)
add_subdirectory(external/SDL_ttf)

set(TH06_SOURCES
    src/AnmManager.cpp
    src/AsciiManager.cpp
    src/BombData.cpp
    src/BulletData.cpp
    src/BulletManager.cpp
    src/Chain.cpp
    src/Controller.cpp
    src/EclManager.cpp
    src/EffectManager.cpp
    src/Ending.cpp
    src/EnemyEclInstr.cpp
    src/EnemyManager.cpp
    src/FileSystem.cpp
    src/GameErrorContext.cpp
    src/GameManager.cpp
    src/GameWindow.cpp
    src/Gui.cpp
    src/ItemManager.cpp
    src/main.cpp
    src/MainMenu.cpp
    src/MidiOutput.cpp
    src/MusicRoom.cpp
    src/Player.cpp
    src/ReplayManager.cpp
    src/ResultScreen.cpp
    src/Rng.cpp
    src/ScreenEffect.cpp
    src/SoundPlayer.cpp
    src/Stage.cpp
    src/Supervisor.cpp
    src/TextHelper.cpp
    src/ZunMath.cpp
    src/utils.cpp
    src/ZunTimer.cpp
    src/graphics/GLFunc.cpp
    src/graphics/FixedFunctionGL.cpp
    src/graphics/WebGL.cpp
    src/graphics/Software.cpp
    src/pbg3/FileAbstraction.cpp
    src/pbg3/IPbg3Parser.cpp
    src/pbg3/Pbg3Archive.cpp
    src/pbg3/Pbg3Parser.cpp
    src/thirdparty/sjis_converter.cpp
    src/midi/MidiDefault.cpp
)

add_executable(th06 ${TH06_SOURCES})

target_include_directories(th06 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)

target_compile_definitions(th06 PRIVATE NDEBUG)
target_compile_options(th06 PRIVATE -Wno-deprecated-declarations)

target_link_libraries(th06 PRIVATE
    SDL2::SDL2-static
    SDL2_image
    SDL2_ttf
)

file(MAKE_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/src/autogenerated)

add_custom_command(
    OUTPUT ${CMAKE_CURRENT_SOURCE_DIR}/src/autogenerated/Embeds.hpp
    COMMAND ${CMAKE_COMMAND}
        -DINPUT_VERT=${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert
        -DINPUT_FRAG=${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag
        -DOUTPUT=${CMAKE_CURRENT_SOURCE_DIR}/src/autogenerated/Embeds.hpp
        -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_shaders.cmake
    DEPENDS
        src/graphics/ff.vert
        src/graphics/ff.frag
        cmake/embed_shaders.cmake
)

add_custom_target(generate_embeds
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/autogenerated/Embeds.hpp
)
add_dependencies(th06 generate_embeds)

set_target_properties(th06 PROPERTIES
    MACOSX_BUNDLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.example.th06"
    MACOSX_BUNDLE_BUNDLE_NAME "Touhou Scarlet Devil"
    MACOSX_BUNDLE_BUNDLE_VERSION "1"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "1.0"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.example.th06"
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1"
    XCODE_ATTRIBUTE_INFOPLIST_KEY_CFBundleDisplayName "東方紅魔郷"
    XCODE_ATTRIBUTE_INFOPLIST_KEY_LSRequiresIPhoneOS "YES"
)
