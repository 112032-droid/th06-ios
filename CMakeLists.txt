cmake_minimum_required(VERSION 3.24)

project(th06 LANGUAGES C CXX OBJC)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

message(STATUS "========================================")
message(STATUS "Building Touhou th06 for iOS")
message(STATUS "Architecture: ${CMAKE_OSX_ARCHITECTURES}")
message(STATUS "SDK: ${CMAKE_OSX_SYSROOT}")
message(STATUS "========================================")

# ============================================================
# SDL2
# ============================================================

set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL2_SHARED OFF CACHE BOOL "" FORCE)

add_subdirectory(
    external/SDL
    ${CMAKE_BINARY_DIR}/SDL
)

# ============================================================
# SDL2_image
# ============================================================

set(SDL2IMAGE_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_VENDORED ON CACHE BOOL "" FORCE)

set(SDL2IMAGE_AVIF OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_WEBP OFF CACHE BOOL "" FORCE)
set(SDL2IMAGE_JXL OFF CACHE BOOL "" FORCE)

add_subdirectory(
    external/SDL_image
    ${CMAKE_BINARY_DIR}/SDL_image
)

# ============================================================
# SDL2_ttf
# ============================================================

set(SDL2TTF_SAMPLES OFF CACHE BOOL "" FORCE)
set(SDL2TTF_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2TTF_VENDORED ON CACHE BOOL "" FORCE)

add_subdirectory(
    external/SDL_ttf
    ${CMAKE_BINARY_DIR}/SDL_ttf
)

# ============================================================
# th06 source files
# ============================================================

set(TH06_SOURCES

    src/main.cpp

    # graphics
    src/graphics/GLFunc.cpp
    src/graphics/FixedFunctionGL.cpp
    src/graphics/WebGL.cpp
    src/graphics/Software.cpp

    # pbg3
    src/pbg3/Background.cpp
    src/pbg3/Camera.cpp
    src/pbg3/Collision.cpp
    src/pbg3/Effect.cpp
    src/pbg3/EffectManager.cpp
    src/pbg3/Enemy.cpp
    src/pbg3/EnemyManager.cpp
    src/pbg3/Font.cpp
    src/pbg3/Game.cpp
    src/pbg3/Item.cpp
    src/pbg3/ItemManager.cpp
    src/pbg3/Math.cpp
    src/pbg3/Player.cpp
    src/pbg3/PlayerShot.cpp
    src/pbg3/PlayerShotManager.cpp
    src/pbg3/Replay.cpp
    src/pbg3/Resource.cpp
    src/pbg3/ResourceManager.cpp
    src/pbg3/Scene.cpp
    src/pbg3/Script.cpp
    src/pbg3/Sound.cpp
    src/pbg3/Stage.cpp
    src/pbg3/Texture.cpp
    src/pbg3/Timer.cpp
    src/pbg3/Util.cpp

    # third party
    src/thirdparty/sjis_converter.cpp

    # MIDI
    src/midi/MidiDefault.cpp
)

# ============================================================
# Automatically remove source files that don't exist
# ============================================================

set(REAL_TH06_SOURCES "")

foreach(SOURCE_FILE ${TH06_SOURCES})

    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${SOURCE_FILE}")

        list(APPEND REAL_TH06_SOURCES "${SOURCE_FILE}")

    else()

        message(WARNING
            "Source file does not exist, skipping: ${SOURCE_FILE}"
        )

    endif()

endforeach()

# ============================================================
# Executable
# ============================================================

add_executable(
    th06
    ${REAL_TH06_SOURCES}
)

target_include_directories(
    th06
    PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src
)

target_compile_definitions(
    th06
    PRIVATE
    NDEBUG
)

target_compile_options(
    th06
    PRIVATE
    -Wno-deprecated-declarations
)

# ============================================================
# Libraries
# ============================================================

target_link_libraries(
    th06
    PRIVATE
    SDL2::SDL2-static
    SDL2_image
    SDL2_ttf
)

# ============================================================
# iOS frameworks
# ============================================================

if(IOS)

    target_link_libraries(
        th06
        PRIVATE
        "-framework Foundation"
        "-framework UIKit"
        "-framework AudioToolbox"
        "-framework AVFoundation"
        "-framework CoreAudio"
        "-framework CoreGraphics"
        "-framework CoreMotion"
        "-framework GameController"
        "-framework Metal"
        "-framework OpenGLES"
        "-framework QuartzCore"
    )

endif()

# ============================================================
# Shader embedding
# ============================================================

file(MAKE_DIRECTORY
    "${CMAKE_CURRENT_SOURCE_DIR}/src/autogenerated"
)

set(EMBED_HEADER
    "${CMAKE_CURRENT_SOURCE_DIR}/src/autogenerated/Embeds.hpp"
)

if(
    EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert"
    AND
    EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag"
)

    add_custom_command(
        OUTPUT
            "${EMBED_HEADER}"

        COMMAND
            ${CMAKE_COMMAND}
            -DINPUT_VERT="${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert"
            -DINPUT_FRAG="${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag"
            -DOUTPUT="${EMBED_HEADER}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_shaders.cmake"

        DEPENDS
            "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.vert"
            "${CMAKE_CURRENT_SOURCE_DIR}/src/graphics/ff.frag"
            "${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed_shaders.cmake"

        VERBATIM
    )

    add_custom_target(
        generate_embeds
        DEPENDS "${EMBED_HEADER}"
    )

    add_dependencies(
        th06
        generate_embeds
    )

endif()

# ============================================================
# iOS application bundle
# ============================================================

set_target_properties(
    th06
    PROPERTIES

    MACOSX_BUNDLE TRUE

    MACOSX_BUNDLE_GUI_IDENTIFIER
        "com.example.th06"

    MACOSX_BUNDLE_BUNDLE_NAME
        "Touhou Scarlet Devil"

    MACOSX_BUNDLE_BUNDLE_VERSION
        "1"

    MACOSX_BUNDLE_SHORT_VERSION_STRING
        "1.0"

    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER
        "com.example.th06"

    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY
        "1"

    XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED
        "NO"

    XCODE_ATTRIBUTE_CODE_SIGNING_REQUIRED
        "NO"

    XCODE_ATTRIBUTE_CODE_SIGN_IDENTITY
        ""

    XCODE_ATTRIBUTE_INFOPLIST_KEY_CFBundleDisplayName
        "東方紅魔郷"

    XCODE_ATTRIBUTE_INFOPLIST_KEY_LSRequiresIPhoneOS
        "YES"
)

message(STATUS "========================================")
message(STATUS "th06 CMake configuration complete")
message(STATUS "System: ${CMAKE_SYSTEM_NAME}")
message(STATUS "Architecture: ${CMAKE_OSX_ARCHITECTURES}")
message(STATUS "SDK: ${CMAKE_OSX_SYSROOT}")

message(STATUS "========================================")
